!VIOS L3 SWITCH 2
!
! 1. FORCE SHUTDOWN 
enable
configure terminal
interface range gigabitEthernet 0/0-3, gigabitEthernet 1/0-3
shutdown 
exit
!
! 2. BASIC TROUBLESHOOTING 
vtp mode transparent
spanning-tree mode rapid-pvst
!
! 3. ALIAS CONFIGURATION 
alias exec seths show etherchannel summary
alias exec sinb show ip interface brief
alias exec sis show interface status
alias exec srun show running-config
alias exec svrb show vrrp brief
alias exec stp show spanning-tree
alias exec svbr show vlan brief
alias exec svlan show vlan
!
! 4. CONSOLE CONFIGURATION 
ip routing
hostname SWL3_2
no ip domain-lookup
line console 0
logging synchronous
exec-timeout 0 0
history size 256
exit
!
! 5. VLAN CONFIGURATION 
vlan 10
name IT
vlan 20
name HR
exit
!
! 6. LACP CONFIGURATION 
!Port-channel-1
interface range gigabitEthernet 0/0-1
description TRUNK TO SW_2
channel-group 1 mode active
no shutdown
!
interface port-channel 1 
switchport
switchport trunk encapsulation dot1q
switchport mode trunk
description LACP TO SW_2
no shutdown
exit
!Port-channel-2
interface range gigabitEthernet 1/0-1
description TRUNK TO SWL3_1 
channel-group 2 mode active
no shutdown
!
interface port-channel 2
switchport
switchport trunk encapsulation dot1q
switchport mode trunk
description LACP TO SWL3_1
no shutdown
exit
!
! 7. VIP AND HSRP
interface vlan 10
description GATEWAY FOR IT VLAN 
ip address 192.168.10.3 255.255.255.0
no shutdown
standby version 2
standby 10 ip 192.168.10.1
standby 10 priority 100
standby 10 preempt 
exit
!
interface vlan 20
description GATEWAY FOR HR VLAN
ip address 192.168.20.3 255.255.255.0
no shutdown
standby version 2
standby 20 ip 192.168.20.1
standby 20 priority 110
standby 20 preempt 
exit
! 8. SAVE CONFIGURATION
end
write memory

!ASA_1
!PASTE IN GLOBAL CONFIGURATION MODE (#)
! 1. CONSOLE CONFIGURATION 
hostname ASA-1
!
! 2. REDUDANT INTERFACE CONFIG
interface gigabitEthernet 0/0
no shutdown
interface gigabitEthernet 0/2
no shutdown
!
interface Redundant 1
member-interface gigabitEthernet 0/0
member-interface gigabitEthernet 0/2
exit
!
! 3. NAMEIF CONFIGURATION
interface redundant 1
nameif inside
security-level 100
ip address 10.0.99.254 255.255.255.0 standby 10.0.99.253
no shutdown
exit 
!
! 4. FAILOVER
interface GigabitEthernet 0/1
description FAILOVER_LINK
no shutdown
exit
!
failover
failover lan unit primary
failover lan interface FAILOVER_LINK GigabitEthernet0/1
failover link FAILOVER_LINK GigabitEthernet0/1
failover interface ip FAILOVER_LINK 192.168.254.1 255.255.255.252 standby 192.168.254.2
!
interface gigabitEthernet0/6
nameif outside
security-level 0
ip address 10.0.0.42 255.255.255.0 standby 10.0.0.43
no shutdown 
exit
!
! 5. ROUTING
route inside 192.168.10.0 255.255.255.0 10.0.99.1
route inside 192.168.20.0 255.255.255.0 10.0.99.1
route inside 192.168.30.0 255.255.255.0 10.0.99.1
route inside 192.168.40.0 255.255.255.0 10.0.99.1
!
! 6. NAT 
object network OBJ-INSIDE-NETS
subnet 192.168.0.0 255.255.0.0
nat (inside,outside) source dynamic OBJ-INSIDE-NETS interface
exit
!

!

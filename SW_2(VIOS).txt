shutdown 
exit
!
! 2. BASIC SETTINGS
username emg_sw2 privilege 15 algorithm-type scrypt secret <REDACTED>
enable algorithm-type scrypt secret <REDACTED>
!

spanning-tree mode rapid-pvst
banner motd #

! 3. ALIAS CONFIGURATION 
alias exec sarch show archive log config all
alias exec seths show etherchannel summary
alias exec sis show interface status
alias exec sit show interfaces trunk
alias exec srun show running-config
alias exec stp show spanning-tree
alias exec svbr show vlan brief
alias exec svlan show vlan
!
! 4. CONSOLE CONFIGURATION 
hostname SW_2
no ip domain-lookup
line console 0
logging synchronous
exec-timeout 0 0
history size 256
exit
!
! 5. VLAN CONFIGURATION 
vlan 20
name HR
!
exit
!
! 6. LACP CONFIGURATION 
!Port-channel-1
interface range gigabitEthernet 0/0-1
description TRUNK_TO_SWL3_2
channel-group 1 mode active
no shutdown
exit
!
interface port-channel 1 
switchport
switchport trunk encapsulation dot1q
switchport mode trunk
switchport trunk allowed vlan 20
description LACP_TO_SWL3_2
no shutdown
exit
!
!Port-channel-4
interface range gigabitEthernet 1/0-1
description TRUNK_TO_SWL3_1
channel-group 4 mode active
no shutdown
exit
!
interface port-channel 4 
switchport
switchport trunk encapsulation dot1q
switchport mode trunk
switchport trunk allowed vlan 20
description LACP_TO_SWL3_1
no shutdown
exit
!
! 7. ACCESS VLAN CONFIGURATION 	
interface range gigabitEthernet 0/2-3
description LINK_TO_VLAN_HR
switchport
switchport mode access
switchport access vlan 20
no shutdown
exit
!
! 8. SPANNING TREE AND PORT SECURITY
interface range gigabitEthernet 0/2-3
spanning-tree portfast
spanning-tree bpduguard enable
switchport port-security
switchport port-security maximum 1
switchport port-security mac-address sticky
switchport port-security violation restrict
exit
!
! 9. VIP REMOTE MANAGEMENT INT CONFIGURATION
interface vlan 20
description MANAGEMENT
ip address 192.168.20.4 255.255.255.0
no shutdown
exit
ip default-gateway 192.168.20.1
!
! 10. AAA RADIUS
aaa new-model
aaa authentication login default group radius local
aaa authorization console
aaa authorization exec default group radius local
aaa accounting exec default start-stop group radius
login on-success log
!
radius server ALPINE_RADIUS
address ipv4 192.168.40.20 auth-port 1812 acct-port 1813
key <REDACTED>
!
ip radius source-interface Vlan 20
!
line vty 0 4
login authentication default
authorization exec default
!
! 11. LOGGING
archive

logging enable
logging size 200
hidekeys
!
! 12. SAVE CONFIGURATION
end

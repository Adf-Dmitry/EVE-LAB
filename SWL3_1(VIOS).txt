!VIOS L3 SWITCH 1
!
! 1. FORCE SHUTDOWN 
enable
configure terminal
interface range gigabitEthernet 0/0-3, gigabitEthernet 1/0-3, gigabitEthernet 2/0-3, gigabitEthernet 3/0-3
shutdown 
exit
!
! 2. BASIC TROUBLESHOOTING 
vtp mode transparent
spanning-tree mode rapid-pvst
!
! 3. ALIAS CONFIGURATION 
alias exec seths show etherchannel summary
alias exec sinb show ip interface brief
alias exec sis show interface status
alias exec srun show running-config
alias exec stbr show standby brief  
alias exec stp show spanning-tree
alias exec svrb show vrrp brief
alias exec svbr show vlan brief
alias exec sro show ip route
alias exec pi8 ping 8.8.8.8
alias exec svlan show vlan
!
! 4. CONSOLE CONFIGURATION 
ip routing
hostname SWL3_1
no ip domain-lookup
line console 0
logging synchronous
exec-timeout 0 0
history size 256
exit
!
! 5. VLAN CONFIGURATION 
vlan 10
name IT
!
vlan 20
name HR
!
vlan 30 
name ADMIN
!
vlan 40
name SRVFARM
!
vlan 99
name TRANSIT
exit
!
! 6. LACP CONFIGURATION 
!Port-channel-1
interface range gigabitEthernet 0/0-1
description TRUNK_TO_SW_1 
channel-group 1 mode active
no shutdown
exit
!
interface port-channel 1 ! GOOD
switchport
switchport trunk encapsulation dot1q
switchport mode trunk
spanning-tree guard root
description LACP_TO_SW_1
no shutdown
exit
!
!Port-channel-2 ! GOOD
interface range gigabitEthernet 1/0-1
description TRUNK_TO_SWL3_2 
channel-group 2 mode active
no shutdown
exit
!
interface port-channel 2
switchport
switchport trunk encapsulation dot1q
switchport mode trunk
description LACP_TO_SWL3_2
no shutdown
exit
!
!Port-channel-3 ! GOOD
interface range gigabitEthernet 2/0-1
description TRUNK_TO_SW_3
channel-group 3 mode active
no shutdown
exit
!
interface port-channel 3
switchport
switchport trunk encapsulation dot1q
switchport mode trunk
spanning-tree guard root
description LACP_TO_SW_3
no shutdown
exit
!
!Port-channel-4 ! GOOD
interface range gigabitEthernet 0/2-3
description TRUNK_TO_SW_2
channel-group 4 mode active
no shutdown
exit
!
interface port-channel 4
switchport
switchport trunk encapsulation dot1q
switchport mode trunk
spanning-tree guard root
description LACP_TO_SW_2
no shutdown
exit
!
!Port-channel-5 ! GOOD
interface range gigabitEthernet 2/2-3
description TRUNK_TO_SW_4 
channel-group 5 mode active
no shutdown
exit
!
interface port-channel 5
switchport
switchport trunk encapsulation dot1q
switchport mode trunk
spanning-tree guard root
description LACP_TO_SW_4
no shutdown
exit
!
!7. ACCESS VLAN CONFIG
interface GigabitEthernet 3/0
description LINK_TO_ASA_1
switchport mode access
switchport access vlan 99
spanning-tree portfast
no shutdown
exit
!
interface GigabitEthernet 3/2
description LINK_TO_ASA_2
switchport mode access
switchport access vlan 99
spanning-tree portfast
no shutdown
exit
!
! 8. VIP AND HSRP
interface vlan 10
description GATEWAY_FOR_IT_VLAN 
ip address 192.168.10.2 255.255.255.0
no shutdown
standby version 2
standby 10 ip 192.168.10.1
standby 10 priority 110
standby 10 preempt 
exit
!
interface vlan 20
description GATEWAY_FOR_HR_VLAN
ip address 192.168.20.2 255.255.255.0
no shutdown
standby version 2
standby 20 ip 192.168.20.1
standby 20 priority 110
standby 20 preempt 
exit
!
interface vlan 30
description GATEWAY_FOR_ADMIN_VLAN
ip address 192.168.30.2 255.255.255.0
no shutdown
standby version 2
standby 30 ip 192.168.30.1
standby 30 priority 100
standby 30 preempt 
exit
!
interface vlan 40
description GATEWAY_FOR_SRV_FARM_VLAN
ip address 192.168.40.2 255.255.255.0
no shutdown
standby version 2
standby 40 ip 192.168.40.1
standby 40 priority 100
standby 40 preempt 
exit
!
interface vlan 99
description TRANSIT_TO_ASA
ip address 10.0.99.2 255.255.255.0
standby version 2
standby 99 ip 10.0.99.1
standby 99 priority 110 
standby 99 preempt 
no shutdown
exit
!
!8. IP ROUTING 
ip route 0.0.0.0 0.0.0.0 10.0.99.254
!
!9. SPANNING TREE (PVST)
spanning-tree vlan 10,20,99 root primary
spanning-tree vlan 30,40 root secondary
end
!
!10. SAVE CONFIGURATION
write memory
!
!
!
!COMPLETE

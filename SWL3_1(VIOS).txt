!VIOS L3 SWITCH 1
!
! 1. FORCE SHUTDOWN 
enable
configure terminal
interface range gigabitEthernet 0/0-3, gigabitEthernet 1/0-3, gigabitEthernet 2/0-3, gigabitEthernet 3/0-3
shutdown 
exit
!
! 2. BASIC TROUBLESHOOTING 
vtp mode transparent
spanning-tree mode rapid-pvst
!
! 3. ALIAS CONFIGURATION 
alias exec seths show etherchannel summary
alias exec sinb show ip interface brief
alias exec sis show interface status
alias exec srun show running-config
alias exec svrb show vrrp brief
alias exec stp show spanning-tree
alias exec svbr show vlan brief
alias exec svlan show vlan
!
! 4. CONSOLE CONFIGURATION 
ip routing
hostname SWL3_1
no ip domain-lookup
line console 0
logging synchronous
exec-timeout 0 0
history size 256
exit
!
! 5. VLAN CONFIGURATION 
vlan 10
name IT
!
vlan 20
name HR
!
vlan 30
name Admin
!
vlan 40
name SrvFarm
exit
!
! 6. LACP CONFIGURATION 
!Port-channel-1
interface range gigabitEthernet 0/0-1
description TRUNK TO SW_1 
channel-group 1 mode active
no shutdown
!
interface port-channel 1 
switchport
switchport trunk encapsulation dot1q
switchport mode trunk
spanning-tree guard root
description LACP TO SW_1
no shutdown
exit
!

! 7. VIP AND HSRP
interface vlan 10
description GATEWAY FOR IT VLAN 
ip address 192.168.10.2 255.255.255.0
no shutdown
standby version 2
standby 10 ip 192.168.10.1
standby 10 priority 110
standby 10 preempt 
exit
!
interface vlan 20
description GATEWAY FOR HR VLAN
ip address 192.168.20.2 255.255.255.0
no shutdown
standby version 2
standby 20 ip 192.168.20.1
standby 20 priority 110
standby 20 preempt 
exit
!
interface vlan 30
description GATEWAY FOR ADMIN VLAN
ip address 192.168.30.2 255.255.255.0
no shutdown
standby version 2
standby 30 ip 192.168.30.1
standby 30 priority 100
standby 30 preempt 
exit
!
interface vlan 40
description GATEWAY FOR SRV_FARM VLAN
ip address 192.168.40.2 255.255.255.0
no shutdown
standby version 2
standby 40 ip 192.168.40.1
standby 40 priority 100
standby 40 preempt 
exit
!
!8. SPANNING TREE (PVST)
spanning-tree vlan 10,20 root primary
spanning-tree vlan 30,40 root secondary
end
!
!9. SAVE CONFIGURATION
write memory
